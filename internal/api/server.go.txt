package api

import (
	"log"
	handler "time_of_armies/internal/app/hadler"
	"time_of_armies/internal/app/repository"

	"github.com/gin-gonic/gin"
	"github.com/sirupsen/logrus"
)

func StartServer() {
	log.Println("Server is running!")

	repo, err := repository.NewRepository()
	if err != nil {
		logrus.Error("ошибка инициализации репозитория!")
	}

	handler := handler.NewHandler(repo)

	r := gin.Default()
	r.LoadHTMLGlob("templates/*")
	r.Static("/static", "./resources")

	r.GET("/armies", handler.GetArmies) // выводим все услуги (армии)
	//r.GET("/armies/:id", handler.SearchArmy)        // выводим результат поиска на странице услуг
	//r.GET("/armies/filter/:id", handler.SearchArmy) // выводим результат поиска на странице услуг
	r.GET("/one_army/:id", handler.GetArmy)                // выводим страницу с отдельной услугой (армией)
	r.GET("/cur_travel_time/:ttid", handler.GetTravelTime) // выводим страницу с текущей заявкой

	r.Run()
	log.Println("Server stopped!")
}
